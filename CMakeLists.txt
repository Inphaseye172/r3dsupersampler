cmake_minimum_required(VERSION 3.25)
project(Supersampler VERSION 0.2.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

option(R3DSS_BUILD_TESTS "Build unit tests" ON)
option(R3DSS_BUILD_BENCH "Build benchmark harness" ON)
set(R3DSS_BUILD_GUI "AUTO" CACHE STRING "Build the Qt GUI (ON/OFF/AUTO)")
set(R3DSS_WITH_CUDA "AUTO" CACHE STRING "Build the CUDA resampler (ON/OFF/AUTO)")
set(R3DSS_WITH_R3DSDK "AUTO" CACHE STRING "Build the RED R3D SDK decoder (ON/OFF/AUTO; needs R3DSDK_ROOT)")

include(warnings)

find_package(nlohmann_json CONFIG REQUIRED)
find_package(CLI11 CONFIG REQUIRED)
find_package(Threads REQUIRED)

# ---- CUDA detection -------------------------------------------------------------------------
set(R3DSS_HAVE_CUDA OFF)
if(NOT R3DSS_WITH_CUDA STREQUAL "OFF")
  include(CheckLanguage)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    enable_language(CUDA)
    set(CMAKE_CUDA_STANDARD 20)
    set(CMAKE_CUDA_STANDARD_REQUIRED ON)
    set(R3DSS_HAVE_CUDA ON)
    if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES OR CMAKE_CUDA_ARCHITECTURES STREQUAL "")
      set(CMAKE_CUDA_ARCHITECTURES 75 86 89)
    endif()
    message(STATUS "CUDA resampler: enabled (${CMAKE_CUDA_COMPILER})")
  elseif(R3DSS_WITH_CUDA STREQUAL "ON")
    message(FATAL_ERROR "R3DSS_WITH_CUDA=ON but no CUDA compiler was found")
  else()
    message(STATUS "CUDA resampler: disabled (no CUDA toolkit found); CPU fallback only")
  endif()
endif()

# ---- RED R3D SDK detection ------------------------------------------------------------------
set(R3DSS_HAVE_R3DSDK OFF)
if(NOT R3DSS_WITH_R3DSDK STREQUAL "OFF")
  find_package(R3DSDK QUIET)
  if(R3DSDK_FOUND)
    set(R3DSS_HAVE_R3DSDK ON)
    message(STATUS "RED R3D SDK: enabled (${R3DSDK_LIBRARY}; runtime ${R3DSDK_RUNTIME_DIR})")
  elseif(R3DSS_WITH_R3DSDK STREQUAL "ON")
    message(FATAL_ERROR "R3DSS_WITH_R3DSDK=ON but the RED R3D SDK was not found; set R3DSDK_ROOT")
  else()
    message(STATUS "RED R3D SDK: not found (set R3DSDK_ROOT); REDCODE clips will be recognised but not decoded")
  endif()
endif()

# ---- Qt detection ---------------------------------------------------------------------------
set(R3DSS_HAVE_GUI OFF)
if(NOT R3DSS_BUILD_GUI STREQUAL "OFF")
  find_package(Qt6 COMPONENTS Widgets QUIET)
  if(Qt6_FOUND)
    set(R3DSS_HAVE_GUI ON)
    message(STATUS "Qt GUI: enabled (Qt ${Qt6_VERSION})")
  elseif(R3DSS_BUILD_GUI STREQUAL "ON")
    message(FATAL_ERROR "R3DSS_BUILD_GUI=ON but Qt6 Widgets was not found; set CMAKE_PREFIX_PATH or QT_ROOT")
  else()
    message(STATUS "Qt GUI: disabled (Qt6 not found); building engine, CLI and tests only")
  endif()
endif()

add_subdirectory(src/core)
add_subdirectory(src/codec)
add_subdirectory(src/codec_cdng)
add_subdirectory(src/synth)
add_subdirectory(src/codec_red)
add_subdirectory(src/codec_r3dsdk)
add_subdirectory(src/resample)
if(R3DSS_HAVE_CUDA)
  add_subdirectory(src/resample_cuda)
endif()
add_subdirectory(src/ratecontrol)
add_subdirectory(src/pipeline)
add_subdirectory(src/cli)
if(R3DSS_HAVE_GUI)
  add_subdirectory(src/gui)
endif()
if(R3DSS_BUILD_BENCH)
  add_subdirectory(bench)
endif()
if(R3DSS_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()

# ---- packaging ------------------------------------------------------------------------------
install(FILES README.md LICENSE NOTICE PATENTS.md DESTINATION .)
install(DIRECTORY docs/ DESTINATION docs FILES_MATCHING PATTERN "*.md")
set(CPACK_PACKAGE_NAME "Supersampler")
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_GENERATOR "ZIP")
set(CPACK_PACKAGE_FILE_NAME "Supersampler-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-x64")
include(CPack)
