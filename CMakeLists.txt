cmake_minimum_required(VERSION 3.25)
project(r3dss VERSION 0.2.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Dynamic CRT everywhere (Qt and the RED SDK library are /MD builds).
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

option(R3DSS_BUILD_TESTS "Build unit tests" ON)
option(R3DSS_BUILD_BENCH "Build benchmark harness" ON)
set(R3DSS_BUILD_GUI "AUTO" CACHE STRING "Build the Qt GUI (ON/OFF/AUTO)")
set(R3DSS_WITH_CUDA "AUTO" CACHE STRING "Build the CUDA resampler (ON/OFF/AUTO)")
set(R3DSS_WITH_R3DSDK "AUTO" CACHE STRING "Build the RED R3D SDK decoder (ON/OFF/AUTO; needs R3DSDK_ROOT)")
option(R3DSS_RELEASE "Production release build: configure fails unless every capability advertised in release/capabilities.json, the verified samples, the GUI and the third-party terms acknowledgement are present" OFF)
option(R3DSS_THIRD_PARTY_TERMS_ACKNOWLEDGED "Set ON only after the maintainer has reviewed the third-party distribution terms listed in docs/REDISTRIBUTION.md" OFF)

include(warnings)

find_package(nlohmann_json CONFIG REQUIRED)
find_package(CLI11 CONFIG REQUIRED)
find_package(Threads REQUIRED)

# ---- CUDA detection -------------------------------------------------------------------------
set(R3DSS_HAVE_CUDA OFF)
if(NOT R3DSS_WITH_CUDA STREQUAL "OFF")
  include(CheckLanguage)
  check_language(CUDA)
  if(CMAKE_CUDA_COMPILER)
    enable_language(CUDA)
    set(CMAKE_CUDA_STANDARD 20)
    set(CMAKE_CUDA_STANDARD_REQUIRED ON)
    set(R3DSS_HAVE_CUDA ON)
    # enable_language() has already filled CMAKE_CUDA_ARCHITECTURES with nvcc's default (e.g. sm_52
    # only), so "still the default" means nobody chose: use Turing / Ampere / Ada, plus Blackwell
    # (RTX 50) when the toolkit knows it. Each entry also embeds PTX for newer GPUs.
    if(CMAKE_CUDA_ARCHITECTURES STREQUAL "" OR CMAKE_CUDA_ARCHITECTURES STREQUAL "${CMAKE_CUDA_ARCHITECTURES_DEFAULT}")
      set(CMAKE_CUDA_ARCHITECTURES 75 86 89)
      if(CMAKE_CUDA_COMPILER_VERSION VERSION_GREATER_EQUAL 12.8)
        list(APPEND CMAKE_CUDA_ARCHITECTURES 120)
      endif()
    endif()
    message(STATUS "CUDA resampler: enabled (${CMAKE_CUDA_COMPILER} ${CMAKE_CUDA_COMPILER_VERSION}; sm ${CMAKE_CUDA_ARCHITECTURES})")
  elseif(R3DSS_WITH_CUDA STREQUAL "ON")
    message(FATAL_ERROR "R3DSS_WITH_CUDA=ON but no CUDA compiler was found")
  else()
    message(STATUS "CUDA resampler: disabled (no CUDA toolkit found); CPU fallback only")
  endif()
endif()

# ---- RED R3D SDK detection ------------------------------------------------------------------
set(R3DSS_HAVE_R3DSDK OFF)
if(NOT R3DSS_WITH_R3DSDK STREQUAL "OFF")
  find_package(R3DSDK QUIET)
  if(R3DSDK_FOUND)
    set(R3DSS_HAVE_R3DSDK ON)
    message(STATUS "RED R3D SDK: enabled (${R3DSDK_LIBRARY}; runtime ${R3DSDK_RUNTIME_DIR})")
  elseif(R3DSS_WITH_R3DSDK STREQUAL "ON")
    message(FATAL_ERROR "R3DSS_WITH_R3DSDK=ON but the RED R3D SDK was not found; set R3DSDK_ROOT")
  else()
    message(STATUS "RED R3D SDK: not found (set R3DSDK_ROOT); REDCODE clips will be recognised but not decoded")
  endif()
endif()

# ---- Qt detection ---------------------------------------------------------------------------
set(R3DSS_HAVE_GUI OFF)
if(NOT R3DSS_BUILD_GUI STREQUAL "OFF")
  find_package(Qt6 COMPONENTS Widgets QUIET)
  if(Qt6_FOUND)
    set(R3DSS_HAVE_GUI ON)
    message(STATUS "Qt GUI: enabled (Qt ${Qt6_VERSION})")
  elseif(R3DSS_BUILD_GUI STREQUAL "ON")
    message(FATAL_ERROR "R3DSS_BUILD_GUI=ON but Qt6 Widgets was not found; set CMAKE_PREFIX_PATH or QT_ROOT")
  else()
    message(STATUS "Qt GUI: disabled (Qt6 not found); building engine, CLI and tests only")
  endif()
endif()

# ---- Release gate ---------------------------------------------------------------------------
# Developer builds (default) degrade gracefully when optional third-party pieces are missing.
# A release must not: it fails here, before anything is built, and scripts/acceptance.py
# re-checks the same manifest against the installed binaries.
if(R3DSS_RELEASE)
  set(_r3dss_missing "")
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/release/capabilities.json" _r3dss_caps)
  string(JSON _r3dss_ncaps LENGTH "${_r3dss_caps}" capabilities)
  math(EXPR _r3dss_last "${_r3dss_ncaps} - 1")
  foreach(_i RANGE ${_r3dss_last})
    string(JSON _r3dss_id GET "${_r3dss_caps}" capabilities ${_i} id)
    string(JSON _r3dss_nreq ERROR_VARIABLE _r3dss_err LENGTH "${_r3dss_caps}" capabilities ${_i} requires)
    if(NOT _r3dss_err AND _r3dss_nreq GREATER 0)
      math(EXPR _r3dss_rlast "${_r3dss_nreq} - 1")
      foreach(_j RANGE ${_r3dss_rlast})
        string(JSON _r3dss_req GET "${_r3dss_caps}" capabilities ${_i} requires ${_j})
        if(_r3dss_req STREQUAL "red-sdk" AND NOT R3DSS_HAVE_R3DSDK)
          list(APPEND _r3dss_missing "capability '${_r3dss_id}' needs the RED R3D SDK (set R3DSDK_ROOT)")
        elseif(NOT _r3dss_req STREQUAL "red-sdk")
          list(APPEND _r3dss_missing "capability '${_r3dss_id}' has an unknown requirement '${_r3dss_req}' (teach CMakeLists.txt about it)")
        endif()
      endforeach()
    endif()
  endforeach()
  if(NOT R3DSS_HAVE_GUI)
    list(APPEND _r3dss_missing "the Qt GUI (Qt6 Widgets not found)")
  endif()
  if(NOT R3DSS_THIRD_PARTY_TERMS_ACKNOWLEDGED)
    list(APPEND _r3dss_missing "R3DSS_THIRD_PARTY_TERMS_ACKNOWLEDGED=ON (maintainer review of docs/REDISTRIBUTION.md)")
  endif()
  find_package(Python3 COMPONENTS Interpreter)
  if(NOT DEFINED ENV{R3DSS_SAMPLES} OR "$ENV{R3DSS_SAMPLES}" STREQUAL "")
    list(APPEND _r3dss_missing "R3DSS_SAMPLES (verified sample clips, see scripts/fetch_samples.py)")
  elseif(NOT Python3_FOUND)
    list(APPEND _r3dss_missing "Python 3 (needed to verify the sample hashes)")
  else()
    execute_process(COMMAND "${Python3_EXECUTABLE}" -I "${CMAKE_CURRENT_SOURCE_DIR}/scripts/fetch_samples.py" --check --dest "$ENV{R3DSS_SAMPLES}"
                    RESULT_VARIABLE _r3dss_samples_rc OUTPUT_VARIABLE _r3dss_samples_out ERROR_VARIABLE _r3dss_samples_out)
    if(NOT _r3dss_samples_rc EQUAL 0)
      list(APPEND _r3dss_missing "verified samples in $ENV{R3DSS_SAMPLES}: ${_r3dss_samples_out}")
    endif()
  endif()
  if(_r3dss_missing)
    list(JOIN _r3dss_missing "\n  - " _r3dss_msg)
    message(FATAL_ERROR "R3DSS_RELEASE=ON but the release is incomplete:\n  - ${_r3dss_msg}")
  endif()
  message(STATUS "Release gate: all advertised capabilities present")
endif()

  foreach(_r3dss_port nlohmann-json cli11)
    set(_r3dss_copyright "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/share/${_r3dss_port}/copyright")
    if(EXISTS "${_r3dss_copyright}")
      install(FILES "${_r3dss_copyright}" DESTINATION licenses RENAME "${_r3dss_port}.txt")
    endif()
  endforeach()
endif()
